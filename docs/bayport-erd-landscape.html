1 — Authentication & security

erDiagram
    users ||--o{ user_roles : has
    roles ||--o{ user_roles : grants
    users ||--o{ mfa_code : receives
    users { bigint id PK varchar username varchar role boolean active }
    roles { bigint id PK varchar name }
    user_roles { bigint user_id FK bigint role_id FK }
    mfa_code { bigint id PK bigint user_id FK varchar code }

    pet_medical_records { bigint id PK bigint pet_id FK varchar record_type date record_date }
